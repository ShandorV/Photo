import io import os import tkinter as tk from tkinter import colorchooser, filedialog, messagebox, ttk from PIL import Image, ImageDraw, ImageTk import segno # Конфігурація шляхів ASSETS_DIR = os.path.join(os.path.dirname(__file__), "assets", "animals") # Дефолтні налаштування fill_color = "#1B4D3E" # Лісовий зелений за замовчуванням back_color = "#FFFFFF" custom_icon_path = None def ensure_assets_dir(): if not os.path.exists(ASSETS_DIR): os.makedirs(ASSETS_DIR, exist_ok=True) def get_preset_animals(): ensure_assets_dir() files = [ f for f in os.listdir(ASSETS_DIR) if f.lower().endswith((".png", ".jpg", ".jpeg", ".webp")) ] return ["Без іконки", "📂 Завантажити свій файл..."] + files def on_preset_change(event): global custom_icon_path selected = combo_presets.get() if selected == "📂 Завантажити свій файл...": path = filedialog.askopenfilename( title="Оберіть зображення тваринки", filetypes=[ ("Зображення", "*.png *.jpg *.jpeg *.webp"), ("Усі файли", "*.*"), ], ) if path: custom_icon_path = path lbl_icon_info.config( text=f"Файл: {os.path.basename(path)}", fg="#2E7D32" ) else: combo_presets.set("Без іконки") custom_icon_path = None lbl_icon_info.config(text="Іконку не обрано", fg="#666666") elif selected == "Без іконки": custom_icon_path = None lbl_icon_info.config(text="Іконку не обрано", fg="#666666") else: custom_icon_path = os.path.join(ASSETS_DIR, selected) lbl_icon_info.config(text=f"Пресет: {selected}", fg="#1976D2") def choose_fill_color(): global fill_color color = colorchooser.askcolor(title="Колір елементів QR", color=fill_color) if color[1]: fill_color = color[1] btn_fill_color.config(bg=fill_color) def choose_back_color(): global back_color color = colorchooser.askcolor(title="Колір фону", color=back_color) if color[1]: back_color = color[1] btn_back_color.config(bg=back_color) def make_corners_rounded(img, radius=15): """Додає м'які закруглені кути для всього QR-коду (ефект стікера)""" mask = Image.new("L", img.size, 0) draw = ImageDraw.Draw(mask) draw.rounded_rectangle([(0, 0), img.size], radius=radius, fill=255) rounded = Image.new("RGBA", img.size, (0, 0, 0, 0)) rounded.paste(img, (0, 0), mask=mask) return rounded def generate_qr(): data = entry_data.get().strip() if not data: messagebox.showwarning("Увага", "Будь ласка, введіть текст або посилання!") return file_path = filedialog.asksaveasfilename( defaultextension=".png", filetypes=[("PNG Image", "*.png"), ("All Files", "*.*")], title="Зберегти кастомний QR-код як...", ) if not file_path: return try: # 1. Генерація базового QR через Segno з високою корекцією помилок (H) qr = segno.make(data, error="h") scale = int(combo_scale.get()) border = int(combo_border.get()) style_shape = combo_style.get() out = io.BytesIO() # Застосування стилів if style_shape == "Округлі точки": qr.save( out, kind="png", dark=fill_color, light=back_color, scale=scale, border=border, ) img = Image.open(out).convert("RGBA") img = make_corners_rounded(img, radius=int(scale * 1.5)) elif style_shape == "М'які кути (Стікер)": qr.save( out, kind="png", dark=fill_color, light=back_color, scale=scale, border=border, ) img = Image.open(out).convert("RGBA") img = make_corners_rounded(img, radius=25) else: qr.save( out, kind="png", dark=fill_color, light=back_color, scale=scale, border=border, ) img = Image.open(out).convert("RGBA") # 2. Накладання іконки тваринки (Пресет або Власний файл) if custom_icon_path and os.path.exists(custom_icon_path): icon = Image.open(custom_icon_path).convert("RGBA") qr_w, qr_h = img.size max_icon_size = int(qr_w * 0.23) icon.thumbnail((max_icon_size, max_icon_size), Image.Resampling.LANCZOS) icon_w, icon_h = icon.size pos_x = (qr_w - icon_w) // 2 pos_y = (qr_h - icon_h) // 2 # Кругла біла підкладка під іконку для стабільного сканування pad = 6 bg_circle = Image.new( "RGBA", (icon_w + pad * 2, icon_h + pad * 2), (0, 0, 0, 0) ) draw_bg = ImageDraw.Draw(bg_circle) draw_bg.ellipse([(0, 0), bg_circle.size], fill=back_color) img.paste(bg_circle, (pos_x - pad, pos_y - pad), mask=bg_circle) img.paste(icon, (pos_x, pos_y), mask=icon) # 3. Збереження файлу img.save(file_path) # 4. Прев'ю у вікні програми img_resized = img.resize((190, 190), Image.Resampling.LANCZOS) img_tk = ImageTk.PhotoImage(img_resized) qr_label.config(image=img_tk) qr_label.image = img_tk status_label.config(text="Успішно збережено!", fg="green") except Exception as e: messagebox.showerror("Помилка", f"Не вдалося створити QR-код:\n{e}") # --- ОБРОБКА БУФЕРА ОБМІНУ ТА ГАРЯЧИХ КЛАВІШ --- def paste_text(event=None): try: text = root.clipboard_get() entry_data.insert(tk.INSERT, text) except tk.TclError: pass return "break" def universal_paste(event): # event.state & 4 перевіряє затиснутий Ctrl # event.keycode == 86 - фізична клавіша 'V' (незалежно від мови розкладки) if (event.state & 4) and event.keycode == 86: return paste_text() def show_context_menu(event): context_menu.tk_popup(event.x_root, event.y_root) # --- ПАРАМЕТРИ ВІКНА --- root = tk.Tk() root.title("Zoo QR Designer") root.geometry("440x700") root.config(padx=15, pady=10) # --- ПОЛЕ ВВЕДЕННЯ --- title_label = tk.Label( root, text="Введіть дані для QR-коду:", font=("Arial", 10, "bold") ) title_label.pack(anchor="w") entry_data = tk.Entry(root, width=48, font=("Arial", 10)) entry_data.pack(pady=(2, 10)) # Прив'язуємо гарячі клавіші та контекстне меню entry_data.bind("", universal_paste) context_menu = tk.Menu(root, tearoff=0) context_menu.add_command(label="Вставити", command=paste_text) entry_data.bind("", show_context_menu) # --- БЛОК 1: ПРЕСЕТИ ТВАРИНОК --- frame_presets = tk.LabelFrame( root, text=" 🦁 Іконка тваринки (Центр) ", font=("Arial", 10, "bold"), padx=10, pady=8, ) frame_presets.pack(fill="x", pady=5) combo_presets = ttk.Combobox( frame_presets, values=get_preset_animals(), state="readonly", font=("Arial", 9), ) combo_presets.set("Без іконки") combo_presets.pack(fill="x", pady=2) combo_presets.bind("<>", on_preset_change) lbl_icon_info = tk.Label( frame_presets, text="Іконку не обрано", font=("Arial", 8), fg="#666666" ) lbl_icon_info.pack(anchor="w", pady=(2, 0)) # --- БЛОК 2: СТИЛІЗАЦІЯ ТА ФОРМА --- frame_style = tk.LabelFrame( root, text=" 🎨 Стилізація та Форма ", font=("Arial", 10, "bold"), padx=10, pady=8, ) frame_style.pack(fill="x", pady=5) # Форма модулів tk.Label(frame_style, text="Стиль елементів:").grid( row=0, column=0, sticky="e", pady=4 ) combo_style = ttk.Combobox( frame_style, values=[ "Стандартний (Квадрати)", "Округлі точки", "М'які кути (Стікер)", ], width=22, state="readonly", ) combo_style.set("Округлі точки") combo_style.grid(row=0, column=1, sticky="w", padx=5, pady=4) # Кольори btn_fill_color = tk.Button( frame_style, text="Колір QR", bg=fill_color, fg="white", font=("Arial", 9, "bold"), width=12, command=choose_fill_color, ) btn_fill_color.grid(row=1, column=0, padx=5, pady=4) btn_back_color = tk.Button( frame_style, text="Колір фону", bg=back_color, fg="black", font=("Arial", 9), width=12, command=choose_back_color, ) btn_back_color.grid(row=1, column=1, padx=5, pady=4) # Розміри tk.Label(frame_style, text="Масштаб (Scale):").grid( row=2, column=0, sticky="e", pady=4 ) combo_scale = ttk.Combobox( frame_style, values=[6, 8, 10, 12, 15], width=6, state="readonly" ) combo_scale.set(10) combo_scale.grid(row=2, column=1, sticky="w", padx=5, pady=4) tk.Label(frame_style, text="Рамка (Border):").grid( row=3, column=0, sticky="e", pady=4 ) combo_border = ttk.Combobox( frame_style, values=[1, 2, 3, 4], width=6, state="readonly" ) combo_border.set(2) combo_border.grid(row=3, column=1, sticky="w", padx=5, pady=4) # --- ГЕНЕРАЦІЯ ТА СТАТУС --- generate_btn = tk.Button( root, text="Згенерувати та зберегти", command=generate_qr, font=("Arial", 11, "bold"), bg="#2E7D32", fg="white", pady=6, ) generate_btn.pack(fill="x", pady=10) status_label = tk.Label(root, text="", font=("Arial", 9)) status_label.pack() qr_label = tk.Label(root) qr_label.pack(pady=5) root.mainloop()
